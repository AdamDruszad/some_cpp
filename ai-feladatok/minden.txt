void cserepointer(int* a, int* b){

void tombkiir(int t[5]){

int minimum(int* t, int meret){
    int min = t[0];
    for(int i = 1; i < meret; i++){
        if(t[i] < min){
            min = t[i];
        }
    }
    return min;
}

int fa(int n){
    int m = 1;
    for(int i = 1; i <= n; i++){
        m = m * i;
    }
    return m;
}

int fakt(int n){
    if(n > 1){
        return n * fakt(n-1);
    }else{
        return 1;
    }
}

int fibo(int n){
    if(n <= 1){
        return 1;
    }else {
        return fibo(n-1) + fibo(n-2);
    }
}

bool isPrime(int n) {
    if (n <= 1) return false;
    for (int i = 2; i <= sqrt(n); i++) {
        if (n % i == 0) return false;
    }
    return true;
}

    }
    if(n == 1){
        return 1;
    }
    return fibo1(n-1)+fibo1(n-2);

int bigfibo(int a, int b) {
    if (b < 0) return -1;

    int legnagyobb_fibo = -1;

    int elozo = 0;
    int jelenlegi = 1;
    

    if (a <= 0 && b >= 0) {
        legnagyobb_fibo = 0;
    }

    while (jelenlegi <= b) {
        if (jelenlegi >= a) {
            legnagyobb_fibo = jelenlegi; 
        }
        
        int kovetkezo = elozo + jelenlegi;
        elozo = jelenlegi;
        jelenlegi = kovetkezo;
    }

    return legnagyobb_fibo;
}
