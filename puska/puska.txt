#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

vector<string> unio(vector<string> a, vector<string> b){
    vector<string> uj;
    for(string i : a){
        uj.push_back(i);
    }
    for(string i : b){
        uj.push_back(i);
    }
    sort(uj.begin(), uj.end());
    for(int i = 0; i < uj.size()-1; i++){
        if(uj[i] == uj[i+1]){
            uj[i].erase();
        }
    }
    return uj;
}

hét hányadik napján volt (1-es indexeléssel)!

Használj pair visszatérési típust!


Tipp (tétel)

Maximum: indulj v[0]-val és 0-s indexszel, for ciklus 1-től. Ha v[i] > max > frissít.

=======================

#include <iostream>
#include <vector>
using namespace std;

pair<int, int> legjobb_nap(vector<int> v) {
    int max_ar = v[0];
    int max_idx = 0;

    for (int i = 1; i < v.size(); i++) {
        if (v[i] > max_ar) {
            max_ar = v[i];
            max_idx = i;
        }
    }

    return {max_ar, max_idx + 1}; // +1 mert 1-es indexelés
}

int main() {
    vector<int> arfolyam = {4250, 4810, 4670, 5120, 4990, 5340, 5210};

    pair<int, int> eredmeny = legjobb_nap(arfolyam);

    cout << "Legmagasabb árfolyam: " << eredmeny.first << " Ft" << endl;
    cout << "A hét " << eredmeny.second << ". napján" << endl;

#include <iostream>
#include <vector>
using namespace std;

void buborek(vector<int>& v) {
    int n = v.size();
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (v[j] > v[j + 1]) {
                int temp = v[j];
                v[j] = v[j + 1];
                v[j + 1] = temp;
            }
        }
    }
}

void kiir(vector<int>& v) {
    for (int i = 0; i < v.size(); i++) {
        cout << v[i];
        if (i < v.size() - 1) cout << ", ";
    }
    cout << endl;
}

int main() {
    vector<int> jegyek = {67, 45, 88, 32, 91, 55, 78, 23, 84};

    cout << "Rendezés előtt: ";
    kiir(jegyek);

    buborek(jegyek);

    cout << "Rendezés után:  ";

Bináris keresés - szótár

Közepes


Adott egy rendezett vektor egész számokkal (pl. szótár oldalszámai):

{12, 25, 38, 47, 56, 63, 71, 89, 94, 107}


Írj függvényt bin_keres(vector v, int ker) névvel, amely bináris kereséssel megkeresi az elemet és

visszaadja indexét (vagy -1-et)!

Teszteld: keresd a 63-at és a 50-et! Írj kommentet, hogy mikor melyik felet dobja el!


Tipp (tétel)

Bináris: elso=0, utolso=n-1. Amíg elso <= utolso: kozep=(e+u)/2. Ha v[kozep] == ker > return. Ha ker

=========================

#include <iostream>
#include <vector>
using namespace std;

int bin_keres(vector<int> v, int ker) {
    int elso = 0;
    int utolso = v.size() - 1;

    while (elso <= utolso) {
        int kozep = (elso + utolso) / 2;

        if (v[kozep] == ker) {
            return kozep;                  // megtaláltuk
        } else if (ker < v[kozep]) {
            utolso = kozep - 1;            // bal felet tartjuk, jobb felet dobjuk el
        } else {
            elso = kozep + 1;              // jobb felet tartjuk, bal felet dobjuk el
        }
    }

    return -1; // nem található
}

int main() {
    vector<int> szótar = {12, 25, 38, 47, 56, 63, 71, 89, 94, 107};

    int ker1 = 63;
    int eredmeny1 = bin_keres(szótar, ker1);
    if (eredmeny1 != -1)
        cout << ker1 << " megtalálva, index: " << eredmeny1 << endl;
    else
        cout << ker1 << " nem található" << endl;

    int ker2 = 50;
    int eredmeny2 = bin_keres(szótar, ker2);
    if (eredmeny2 != -1)
        cout << ker2 << " megtalálva, index: " << eredmeny2 << endl;
    else
        cout << ker2 << " nem található" << endl;

    return 0;
}

Összefuttatás - névsorok

Két rendezett vektor összefésülése


Nehéz


Adott két rendezett vektor (pl. két osztály névsorának betűrendes kódjai):

A = {11, 23, 34, 45, 67}

B = {15, 23, 41, 67, 89, 95}


Írj függvényt osszefuttatas(vector a, vector b) névvel, amely rendezetten fésüli össze a két

vektort! (Ha mindkettőben szerepel egy szám, csak egyszer szerepeljen az eredményben!)


Tipp (tétel)

Összefuttatás: i=0, j=0 mutatók. Amíg mindkettő él: kisebbet tedd be, azt a mutatót léptesd. Végén

a maradékot told be.

========================

#include <iostream>
#include <vector>
using namespace std;

vector<int> osszefuttatas(vector<int> a, vector<int> b) {
    vector<int> eredmeny;
    int i = 0, j = 0;

    while (i < a.size() && j < b.size()) {
        if (a[i] < b[j]) {
            eredmeny.push_back(a[i]);
            i++;
        } else if (a[i] > b[j]) {
            eredmeny.push_back(b[j]);
            j++;
        } else {
            // egyenlők → csak egyszer tesszük be
            eredmeny.push_back(a[i]);
            i++;
            j++;
        }
    }

    // A maradék betolása
    while (i < a.size()) {
        eredmeny.push_back(a[i]);
        i++;
    }
    while (j < b.size()) {
        eredmeny.push_back(b[j]);
        j++;
    }

    return eredmeny;
}

void kiir(vector<int>& v) {
    for (int i = 0; i < v.size(); i++) {
        cout << v[i];
        if (i < v.size() - 1) cout << ", ";
    }
    cout << endl;
}

int main() {
    vector<int> A = {11, 23, 34, 45, 67};
    vector<int> B = {15, 23, 41, 67, 89, 95};

    cout << "A: "; kiir(A);
    cout << "B: "; kiir(B);

    vector<int> eredmeny = osszefuttatas(A, B);
    cout << "Összefésülve: "; kiir(eredmeny);

    return 0;
}

Rekurzió - hatványozás

Rekurzív függvény írása


Közepes


Írj rekurzív függvényt hatvany(int alap, int kitevo) névvel, amely kiszámítja az alap kitevo-adik

hatványát!

(Tehát: hatvany(2, 10) = 1024)


Feltételek:

- NEM használhatsz pow()-t!

- A kitevo mindig nemnegatív egész!

- Írd le a rekurzív összefüggést kommentként!

Teszteld: 2^10, 3^4, 5^0


Tipp (tétel)

Báziseset: kitevo == 0 > return 1. Rekurzív eset: return alap * hatvany(alap, kitevo-1).

=====================

#include <iostream>
using namespace std;

// Rekurzív összefüggés:
// hatvany(alap, 0)      = 1
// hatvany(alap, kitevo) = alap * hatvany(alap, kitevo - 1)

int hatvany(int alap, int kitevo) {
    if (kitevo == 0) return 1;                        // báziseset
    return alap * hatvany(alap, kitevo - 1);          // rekurzív eset
}

int main() {
    cout << "2^10 = " << hatvany(2, 10) << endl;
    cout << "3^4  = " << hatvany(3, 4)  << endl;

Teszteld: határ=100, intervallum [20, 40]


Tipp (tétel)

#include <iostream>
using namespace std;

void fibo_listaz(int hatar) {
    int elozo = 0, jelenlegi = 1;

    cout << "Fibonacci számok (≤ " << hatar << "): ";
    cout << elozo << ", ";

    while (jelenlegi <= hatar) {
        cout << jelenlegi;
        int kov = elozo + jelenlegi;
        elozo = jelenlegi;
        jelenlegi = kov;
        if (jelenlegi <= hatar) cout << ", ";
    }
    cout << endl;
}

int fibo_kozt(int a, int b) {
    int elozo = 0, jelenlegi = 1;

    if (elozo >= a && elozo <= b) return elozo;

    while (jelenlegi <= b) {
        if (jelenlegi >= a && jelenlegi <= b) return jelenlegi;
        int kov = elozo + jelenlegi;
        elozo = jelenlegi;
        jelenlegi = kov;
    }

    return -1; // nincs ilyen
}

int main() {
    fibo_listaz(100);

    int a = 20, b = 40;
    int eredmeny = fibo_kozt(a, b);

    if (eredmeny != -1)
        cout << "Első Fibonacci [" << a << ", " << b << "] között: " << eredmeny << endl;
    else
        cout << "Nincs Fibonacci szám [" << a << ", " << b << "] között" << endl;

    return 0;

// Rekurzív összefüggés:
// lnko(a, 0) = a
// lnko(a, b) = lnko(b, a % b)

int lnko(int a, int b) {

    return lnko(b, a % b);          // rekurzív eset
}

Medián és átlag - statisztika

Statisztikai alapszámítások


Közepes


Adott egy vektor diákok ZH pontszámaival:

{72, 85, 61, 90, 45, 78, 55, 88, 73, 67}


Írj három függvényt:

1. atlag(vector v) > double, a pontszámok átlaga

2. median(vector v) > double, a medián érték

3. szoras(vector v) > double, a szórás (sqrt(átlag of (xi-átlag)2))


Írasd ki mindhármat!


Tipp (tétel)

Medián: sort(), ha n páros: (v[n/2-1]+v[n/2])/2.0, ha páratlan: v[n/2]. Szórás: #include szükséges.

=======================

#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>
using namespace std;

double atlag(vector<int> v) {
    double osszeg = 0;
    for (int x : v) osszeg += x;
    return osszeg / v.size();
}

double median(vector<int> v) {
    sort(v.begin(), v.end());
    int n = v.size();
    if (n % 2 == 0)
        return (v[n/2 - 1] + v[n/2]) / 2.0;   // páros: két középső átlaga
    else
        return v[n/2];                           // páratlan: középső elem
}

double szoras(vector<int> v) {
    double atl = atlag(v);
    double osszeg = 0;
    for (int x : v) osszeg += (x - atl) * (x - atl);
    return sqrt(osszeg / v.size());
}

int main() {
    vector<int> pontok = {72, 85, 61, 90, 45, 78, 55, 88, 73, 67};

    cout << "Átlag:  " << atlag(pontok)  << endl;
    cout << "Medián: " << median(pontok) << endl;
    cout << "Szórás: " << szoras(pontok) << endl;

    return 0;
}

#include <vector>
using namespace std;

int main() {
    vector<vector<int>> jegyek = {
        {5, 4, 3, 5, 4},
        {3, 2, 4, 3, 5},
        {5, 5, 5, 4, 5},
        {2, 3, 2, 4, 3}
    };

    int sorok   = jegyek.size();
    int oszlopok = jegyek[0].size();

    // 1. Sorátlagok (diákok átlaga)
    cout << "=== Diákok átlaga ===" << endl;
    int legjobb_idx = 0;
    double legjobb_atlag = 0;

        double osszeg = 0;
        for (int j = 0; j < oszlopok; j++) osszeg += jegyek[i][j];
        double atlag = osszeg / oszlopok;
        cout << "Diák " << i+1 << ": " << atlag << endl;

        if (atlag > legjobb_atlag) {
            legjobb_atlag = atlag;
            legjobb_idx = i;
        }
    }

    // 2. Oszlopátlagok (tárgyak átlaga)
    cout << endl << "=== Tárgyak átlaga ===" << endl;
    for (int j = 0; j < oszlopok; j++) {
        double osszeg = 0;
        for (int i = 0; i < sorok; i++) osszeg += jegyek[i][j];
        cout << "Tárgy " << j+1 << ": " << osszeg / sorok << endl;
    }

    // 3. Legjobb diák
    cout << endl << "=== Legjobb diák ===" << endl;
    cout << "Diák " << legjobb_idx+1 << " (átlag: " << legjobb_atlag << ")" << endl;

    // 4. Főátló elemei
    cout << endl << "=== Főátló elemei ===" << endl;
    int atlo_meret = min(sorok, oszlopok);
    for (int i = 0; i < atlo_meret; i++) {
        cout << "jegyek[" << i << "][" << i << "] = " << jegyek[i][i] << endl;
    }

    return 0;

#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cout << "Hány szót adsz meg? ";
    cin >> n;

    vector<string> szavak;
    for (int i = 0; i < n; i++) {
        string s;
        cout << i+1 << ". szó: ";
        cin >> s;
        szavak.push_back(s);
    }

    // 1. Leghosszabb és legrövidebb
    string leghosszabb = szavak[0];
    string legroviebb  = szavak[0];
    double ossz_hossz  = 0;

    for (string s : szavak) {
        ossz_hossz += s.size();
        if (s.size() > leghosszabb.size()) leghosszabb = s;
        if (s.size() < legroviebb.size())  legroviebb  = s;
    }

    double atlag = ossz_hossz / n;

    cout << endl;
    cout << "=== Leghosszabb szó ===" << endl;
    cout << leghosszabb << " (" << leghosszabb.size() << " karakter)" << endl;

    cout << endl << "=== Legrövidebb szó ===" << endl;
    cout << legroviebb << " (" << legroviebb.size() << " karakter)" << endl;

    // 2. Átlagos szóhossz
    cout << endl << "=== Átlagos szóhossz ===" << endl;
    cout << atlag << " karakter" << endl;

    // 3. Átlag feletti szavak
    cout << endl << "=== Átlagnál hosszabb szavak ===" << endl;
    for (string s : szavak) {
        if (s.size() > atlag) cout << s << " (" << s.size() << ")" << endl;
    }

    // 4. Rendezés hossz szerint
    sort(szavak.begin(), szavak.end(), [](string a, string b) {
        return a.size() < b.size();
    });

    cout << endl << "=== Szavak hossz szerint ===" << endl;
    for (string s : szavak) {
        cout << s << " (" << s.size() << ")" << endl;
    }

    return 0;
}
